cmake_minimum_required(VERSION 3.16)
project(railguard_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -march=native -Wall")
find_package(OpenCV REQUIRED)
set(ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/onnxruntime")
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include ${ORT_ROOT}/include ${OpenCV_INCLUDE_DIRS})
link_directories(${ORT_ROOT}/lib)
add_executable(railguard_engine src/main.cpp)
target_link_libraries(railguard_engine onnxruntime ${OpenCV_LIBS} pthread)
